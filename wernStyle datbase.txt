CREATE DATABASE wern_style;


CREATE TABLE users(
    user_id int PRIMARY KEY AUTO_INCREMENT,
    user_name varchar(100) NOT NULL,
    user_email varchar(100) UNIQUE,
    password varchar(50) NOT NULL,
    creatad_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)





CREATE TABLE styles(
    style_id int PRIMARY KEY AUTO_INCREMENT,
    style_name varchar(100),
    discription text, 
    category varchar(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)





CREATE TABLE news_letter(
    news_letter_id int PRIMARY KEY AUTO_INCREMENT,
    email varchar(100) UNIQUE NOT NULL,
    subscribed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)






CREATE TABLE contact_message(
    message_id int PRIMARY KEY AUTO_INCREMENT ,
    contact_name varchar(255) NOT NULL ,
    contact_email varchar(300) UNIQUE NOT NULL,
    contact_messsage text,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)






CREATE TABLE products(
    products_id int PRIMARY KEY AUTO_INCREMENT ,
    title varchar(300) NOT NULL,
    description TEXT,
    price DECIMAL(10,2),
    stock int DEFAULT 0 ,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    style_id int ,
    FOREIGN KEY(style_id)REFERENCES styles(style_id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
)









CREATE TABLE cart (
    cart_id int PRIMARY KEY AUTO_INCREMENT ,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    user_id int,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
)





CREATE TABLE favorites(
    favorites_id int PRIMARY KEY AUTO_INCREMENT,
    user_id int,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
    
    products_id int ,
    FOREIGN KEY (products_id) REFERENCES products(products_id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
)






CREATE TABLE cart_items(
    cart_item_id int PRIMARY KEY AUTO_INCREMENT,
    quantity INT,
    cart_id int,
    FOREIGN key(cart_id) REFERENCES cart(cart_id)
    ON DELETE  CASCADE
    ON UPDATE CASCADE,
    
    products_id int,
    FOREIGN KEY (products_id) REFERENCES products(products_id)
)
